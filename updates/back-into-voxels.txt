func _update_rolling_boulders(delta: float) -> void:
	var max_col_dist := float(terrain.collision_distance_chunks) * 32.0 * terrain.voxel_scale
	for i in range(_rolling_boulders.size() - 1, -1, -1):
		var body := _rolling_boulders[i]
		if not is_instance_valid(body):
			_rolling_boulders.remove_at(i)
			continue
		# Freeze or despawn if it rolled past the active collision chunk radius
		if body.global_position.distance_to(camera.global_position) > max_col_dist:
			body.queue_free()
			_rolling_boulders.remove_at(i)
			continue
		# Optional hybrid voxel-bake once settled
		if body.has_meta("settle_timer"):
			if body.linear_velocity.length() < 0.25 and body.angular_velocity.length() < 0.4:
				var t := float(body.get_meta("settle_timer")) + delta
				body.set_meta("settle_timer", t)
				if t >= 1.2:
					var r := float(body.get_meta("boulder_radius"))
					terrain.place_boulder(body.global_position, r)
					body.queue_free()
					_rolling_boulders.remove_at(i)
			else:
				body.set_meta("settle_timer", 0.0)
