## Connected to terrain.explosion_completed(center: Vector3, radius: float, affected_chunks: int)
func _on_terrain_explosion_physics(center: Vector3, radius: float, _affected_chunks: int) -> void:
	var wake_radius := radius + 6.0
	var blast_radius := radius + 4.0
	for i in range(_rolling_boulders.size() - 1, -1, -1):
		var body := _rolling_boulders[i]
		if not is_instance_valid(body):
			_rolling_boulders.remove_at(i)
			continue
		var offset := body.global_position - center
		var dist := offset.length()
		# 1. Always wake bodies near modified chunks so they drop into fresh craters
		if dist <= wake_radius:
			body.sleeping = false
		# 2. Apply an outward + upward shockwave impulse if caught in the blast
		if dist <= blast_radius and dist > 0.001:
			var falloff := clampf(1.0 - (dist / blast_radius), 0.0, 1.0)
			var dir := (offset / dist + Vector3(0.0, 0.35, 0.0)).normalized()
			body.apply_central_impulse(dir * (body.mass * 14.0 * falloff))
